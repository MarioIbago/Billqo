function todayIso(): string {
  const now = new Date();
  const month = String(now.getMonth() + 1).padStart(2, '0');
  const day = String(now.getDate()).padStart(2, '0');
  return `${now.getFullYear()}-${month}-${day}`;
}

function createClientId(): string {
  if (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function') return crypto.randomUUID();
  return `${Date.now()}-${Math.random().toString(36).slice(2)}`;
}

function formatMoney(amount: number, currency = 'MXN'): string {
  return new Intl.NumberFormat('es-MX', { style: 'currency', currency, maximumFractionDigits: 2 }).format(amount);
}

function initials(user?: User | null): string {
  const source = user?.displayName || user?.email || 'Usuario';
  return source.split(/\s+/).map((part) => part[0]).join('').slice(0, 2).toUpperCase();
}

function displayName(user?: User | null): string {
  return user?.displayName || user?.email?.split('@')[0] || 'Usuario';
}

function monthBounds(): { startDate: string; endDate: string } {
  const now = new Date();
  const startDate = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`;
  const endDate = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-${String(new Date(now.getFullYear(), now.getMonth() + 1, 0).getDate()).padStart(2, '0')}`;
  return { startDate, endDate };
}

function errorMessage(error: unknown): string {
  return error instanceof Error ? error.message : 'No pudimos completar la operación. Inténtalo de nuevo.';
}

function pointsFor(values: number[]): string {
  if (values.length === 0) return '3,28 97,28';
  const max = Math.max(...values, 1);
  return values.map((value, index) => {
    const x = values.length === 1 ? 50 : 3 + (94 * index) / (values.length - 1);
    const y = 28 - (value / max) * 22;
    return `${x.toFixed(1)},${y.toFixed(1)}`;
  }).join(' ');
}

function IncomeExpenseChart({ transactions, currency }: { transactions: Transaction[]; currency: string }) {
  const series = useMemo(() => {
    const dates = [...new Set(transactions.map((transaction) => transaction.date))].sort().slice(-8);
    return dates.map((date) => ({
      date,
      income: transactions.filter((transaction) => transaction.date === date && transaction.type === 'income').reduce((total, transaction) => total + transaction.amount, 0),
      expense: transactions.filter((transaction) => transaction.date === date && transaction.type === 'expense').reduce((total, transaction) => total + transaction.amount, 0),
    }));
  }, [transactions]);

  if (series.length === 0) return <EmptyChart label="Aún no hay movimientos para graficar." />;
  const maximum = Math.max(...series.flatMap((item) => [item.income, item.expense]), 1);
  const pathFor = (key: 'income' | 'expense') => series.map((item, index) => {
    const x = series.length === 1 ? 290 : 14 + (552 * index) / (series.length - 1);
    const y = 182 - (item[key] / maximum) * 150;
    return `${x.toFixed(1)},${y.toFixed(1)}`;
  }).join(' ');
  return (
    <div className="income-chart-wrap">
      <div className="chart-legend"><span><i className="legend-dot green" />Ingresos</span><span><i className="legend-dot red" />Gastos</span><label>Fuente: <b>Google Sheet</b></label></div>
      <svg className="line-chart" viewBox="0 0 580 210" role="img" aria-label="Ingresos y gastos registrados">
        {[32, 70, 108, 146, 184].map((y) => <line key={y} x1="14" x2="566" y1={y} y2={y} className="chart-grid-line" />)}
        <polyline points={pathFor('income')} fill="none" stroke="#0aa83d" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
        <polyline points={pathFor('expense')} fill="none" stroke="#ef4444" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
        <text x="14" y="202">{series[0]?.date.slice(5).replace('-', '/')}</text>
        <text x="500" y="202">{series.at(-1)?.date.slice(5).replace('-', '/')}</text>
        <text x="14" y="21">{formatMoney(maximum, currency)}</text>
      </svg>
    </div>
  );
}

type CategorySlice = { name: string; amount: number; percentage: number; color: string };

function DonutChart({ slices, total, currency, expanded, onToggle }: {
  slices: CategorySlice[];
  total: number;
  currency: string;
  expanded: boolean;
  onToggle: () => void;
}) {
  if (slices.length === 0) return <EmptyChart label="Los gastos por categoría aparecerán aquí." />;
  const radius = 54;
  const circumference = 2 * Math.PI * radius;
  let offset = 0;
  const visible = expanded ? slices : slices.slice(0, 5);
  return (
    <div className="category-chart-content">
      <div className="donut-wrap"><svg className="donut" viewBox="0 0 160 160" role="img" aria-label="Distribución real de gastos por categoría">
        <circle cx="80" cy="80" r={radius} fill="none" stroke="#edf0f2" strokeWidth="22" />
        {slices.map((slice) => {
          const length = circumference * (slice.percentage / 100);
          const currentOffset = offset;
          offset += length;
          return <circle key={slice.name} cx="80" cy="80" r={radius} fill="none" stroke={slice.color} strokeWidth="22" strokeDasharray={`${length} ${circumference - length}`} strokeDashoffset={-currentOffset} transform="rotate(-90 80 80)" />;
        })}
        <text x="80" y="78" textAnchor="middle" className="donut-total">{formatMoney(total, currency)}</text><text x="80" y="95" textAnchor="middle" className="donut-label">Total</text>
      </svg></div>
      <div className="category-legend">
        {visible.map((slice) => <div className="category-row" key={slice.name}><span className="category-name"><i style={{ background: slice.color }} />{slice.name}</span><span>{slice.percentage.toFixed(1)}% <em>({formatMoney(slice.amount, currency)})</em></span></div>)}
        {slices.length > 5 && <button type="button" className="category-toggle" onClick={onToggle}>{expanded ? 'Ocultar detalle' : 'Ver detalle'}</button>}
      </div>
    </div>
  );
}

function TransactionEditor({ transaction, categories, onClose, onSave }: { key?: React.Key;
  transaction?: Transaction;
  categories: Category[];
  onClose: () => void;
  onSave: (payload: TransactionPayload, idempotencyKey: string) => Promise<void>;
}) {
  const initialType: TransactionType = transaction?.type ?? 'expense';
  const startingCategory = categories.find((category) => category.id === transaction?.categoryId) ?? categories.find((category) => category.type === initialType);
  const [type, setType] = useState<TransactionType>(initialType);
  const [description, setDescription] = useState(transaction?.description ?? '');
  const [amount, setAmount] = useState(transaction ? String(transaction.amount) : '');
  const [categoryId, setCategoryId] = useState(startingCategory?.id ?? '');
  const [costType, setCostType] = useState<CostType>(transaction?.costType ?? (initialType === 'income' ? 'Ingreso' : 'Variable'));
  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>(transaction?.paymentMethod ?? 'Tarjeta Débito');
  const [date, setDate] = useState(transaction?.date ?? todayIso());
  const [fixedVariable, setFixedVariable] = useState<FixedVariable>(transaction?.fixedVariable ?? 'Variable');
  const [necessity, setNecessity] = useState<Necessity>(transaction?.necessity ?? 'Necesario');
  const [influence, setInfluence] = useState<1 | 2 | 3 | 4 | 5>(transaction?.influence ?? 3);
  const [account, setAccount] = useState(transaction?.account ?? '');
  const [notes, setNotes] = useState(transaction?.notes ?? '');
  const [tags, setTags] = useState(transaction?.tags?.join(', ') ?? '');
  const [recurring, setRecurring] = useState(Boolean(transaction?.recurring));
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string>();
  const idempotencyKey = useRef(createClientId());

  const typeCategories = categories.filter((category) => category.type === type && category.active);
  const switchType = (nextType: TransactionType) => {
    setType(nextType);
    setCategoryId(categories.find((category) => category.type === nextType && category.active)?.id ?? '');
    setCostType(nextType === 'income' ? 'Ingreso' : 'Variable');
  };

  const submit = async (event: React.FormEvent) => {
    event.preventDefault();
    const category = typeCategories.find((item) => item.id === categoryId);
    if (!category) {
      setError('Selecciona una categoría disponible.');
      return;
    }
    const parsedAmount = Number(amount);
    if (!Number.isFinite(parsedAmount) || parsedAmount <= 0) {
      setError('Ingresa un monto mayor que cero.');
      return;
    }
    setSaving(true);
    setError(undefined);
    try {
      await onSave({
        description,
        amount: parsedAmount,
        type,
        categoryId: category.id,
        category: category.name,
        costType: type === 'income' ? 'Ingreso' : costType,
        fixedVariable: type === 'expense' ? fixedVariable : undefined,
        necessity: type === 'expense' ? necessity : undefined,
        influence: type === 'expense' ? influence : undefined,
        paymentMethod,
        date,
        account: account || undefined,
        notes: notes || undefined,
        tags: tags.split(',').map((tag) => tag.trim()).filter(Boolean),
        recurring,
      }, idempotencyKey.current);
      onClose();
    } catch (caught) {
      setError(errorMessage(caught));
    } finally {
      setSaving(false);
    }
  };

  return <div className="modal-backdrop" role="presentation" onMouseDown={(event) => { if (!saving && event.target === event.currentTarget) onClose(); }}>
    <section className="movement-modal movement-modal-wide" role="dialog" aria-modal="true" aria-labelledby="movement-title">
      <div className="modal-heading"><div><span className="modal-kicker">{transaction ? 'Edición confirmada en Google Sheets' : 'Registro en Google Sheets'}</span><h2 id="movement-title">{transaction ? 'Editar movimiento' : 'Nuevo movimiento'}</h2></div><button type="button" className="modal-close" onClick={onClose} disabled={saving} aria-label="Cerrar"><X size={18} /></button></div>
      <form onSubmit={submit} className="movement-form">
        <div className="movement-type-toggle"><button type="button" className={type === 'expense' ? 'is-active expense' : ''} onClick={() => switchType('expense')} disabled={saving}>Gasto</button><button type="button" className={type === 'income' ? 'is-active income' : ''} onClick={() => switchType('income')} disabled={saving}>Ingreso</button></div>
        {error && <p className="form-error" role="alert">{error}</p>}
        <div className="movement-fields-grid"><label>Concepto<input autoFocus required value={description} onChange={(event) => setDescription(event.target.value)} disabled={saving} maxLength={240} /></label><label>Monto<input required type="number" min="0.01" step="0.01" value={amount} onChange={(event) => setAmount(event.target.value)} disabled={saving} /></label></div>
        <div className="movement-fields-grid"><label>Categoría<select value={categoryId} onChange={(event) => setCategoryId(event.target.value)} disabled={saving}>{typeCategories.map((category) => <option key={category.id} value={category.id}>{category.name}</option>)}</select></label><label>Método de pago<select value={paymentMethod} onChange={(event) => setPaymentMethod(event.target.value as PaymentMethod)} disabled={saving}>{paymentMethods.map((method) => <option key={method} value={method}>{method}</option>)}</select></label></div>
        <div className="movement-fields-grid"><label>Fecha<input type="date" value={date} onChange={(event) => setDate(event.target.value)} disabled={saving} /></label>{type === 'expense' ? <label>Clasificación<select value={costType} onChange={(event) => setCostType(event.target.value as CostType)} disabled={saving}>{expenseCostTypes.map((item) => <option key={item} value={item}>{item}</option>)}</select></label> : <label>Clasificación<input value="Ingreso" disabled /></label>}</div>
        {type === 'expense' && <><div className="movement-fields-grid"><label>Fijo o variable<select value={fixedVariable} onChange={(event) => setFixedVariable(event.target.value as FixedVariable)} disabled={saving}><option value="Fijo">Fijo</option><option value="Variable">Variable</option></select></label><label>Necesidad<select value={necessity} onChange={(event) => setNecessity(event.target.value as Necessity)} disabled={saving}><option value="Necesario">Necesario</option><option value="Innecesario">Innecesario</option></select></label></div><label>Influencia al decidir (1 a 5)<select value={influence} onChange={(event) => setInfluence(Number(event.target.value) as 1 | 2 | 3 | 4 | 5)} disabled={saving}>{[1, 2, 3, 4, 5].map((value) => <option key={value} value={value}>{value}</option>)}</select></label></>}
        <details className="movement-details"><summary>Más detalles</summary><div className="movement-fields-grid"><label>Cuenta<input value={account} onChange={(event) => setAccount(event.target.value)} disabled={saving} maxLength={120} /></label><label>Etiquetas (separadas por coma)<input value={tags} onChange={(event) => setTags(event.target.value)} disabled={saving} /></label></div><label>Notas<textarea value={notes} onChange={(event) => setNotes(event.target.value)} disabled={saving} maxLength={2000} /></label><label className="checkbox-label"><input type="checkbox" checked={recurring} onChange={(event) => setRecurring(event.target.checked)} disabled={saving} /> Guardar como definición recurrente mensual (sin generación automática)</label></details>
        <div className="modal-actions"><button type="button" className="secondary-button" onClick={onClose} disabled={saving}>Cancelar</button><button type="submit" className="primary-button" disabled={saving}>{saving ? <><LoaderCircle className="spin" size={16} />Guardando en Sheets…</> : <><Plus size={16} />{transaction ? 'Guardar cambios' : 'Añadir movimiento'}</>}</button></div>
      </form>
    </section>
  </div>;
}

  categories: Category[];
  budgets: CategoryBudget[];
  preferences: FinancialPreferences;
  onSave: (id: string, categoryId: string, amount: number, expectedUpdatedAt?: string) => Promise<void>;
}) {
  const [drafts, setDrafts] = useState<Record<string, string>>({});
  const [saving, setSaving] = useState<string>();
  const [error, setError] = useState<string>();
  const budgetByCategory = new Map(budgets.filter((budget) => budget.active !== false && !budget.deletedAt).map((budget) => [budget.categoryId, budget]));
  const expenseCategories = categories.filter((category) => category.type === 'expense' && category.active);
  const save = async (category: Category) => {
    const budget = budgetByCategory.get(category.id);
    const amount = Number(drafts[category.id] ?? budget?.allocatedAmount ?? 0);
    if (!Number.isFinite(amount) || amount < 0) { setError('El límite debe ser un número igual o mayor que cero.'); return; }
    setSaving(category.id); setError(undefined);
    try { await onSave(budget?.id ?? createClientId(), category.id, amount, budget?.updatedAt); } catch (caught) { setError(errorMessage(caught)); } finally { setSaving(undefined); }
  };
  return <section className="workspace-panel"><div className="workspace-panel-heading"><div><span className="panel-kicker">PRESUPUESTOS</span><h2>Presupuesto por categoría</h2><p>El gasto acumulado se calcula desde movimientos activos del periodo del archivo.</p></div><strong className="global-budget">Global: {formatMoney(preferences.monthlyBudget, preferences.currency)}</strong></div>{error && <p className="form-error">{error}</p>}<div className="budget-grid">{expenseCategories.map((category) => { const budget = budgetByCategory.get(category.id); const limit = budget?.allocatedAmount ?? 0; const spent = budget?.spentAmount ?? 0; const percentage = limit > 0 ? (spent / limit) * 100 : 0; return <article className="budget-item" key={category.id}><div><strong>{category.name}</strong><small>Gastado: {formatMoney(spent, preferences.currency)}</small></div><div className="budget-input-row"><input type="number" min="0" step="0.01" value={drafts[category.id] ?? String(limit)} onChange={(event) => setDrafts((current) => ({ ...current, [category.id]: event.target.value }))} /><button type="button" onClick={() => void save(category)} disabled={saving === category.id}>{saving === category.id ? '…' : 'Guardar'}</button></div><div className="budget-progress"><i style={{ width: `${Math.min(100, percentage)}%` }} /></div><small>{limit > 0 ? `${percentage.toFixed(1)}% del límite` : 'Sin límite configurado'}</small></article>; })}</div></section>;
}

  connection: GoogleConnection;
  preferences: FinancialPreferences;
  syncedAt: string;
  onSync: () => Promise<void>;
  onSavePreferences: (preferences: Partial<FinancialPreferences>, expectedUpdatedAt: string) => Promise<void>;
  onReconnect: () => Promise<void>;
  onDisconnect: () => Promise<void>;
  onSignOut: () => Promise<void>;

Sincroniza e inténtalo de nuevo.'); await onSavePreferences({ currency, dateFormat, timezone, monthlyBudget: Number(monthlyBudget) }, preferences.updatedAt); setMessage('Configuración guardada en la pestaña CONFIGURACIÓN.'); } catch (caught) { setMessage(errorMessage(caught)); } finally { setBusy(false); } };

  return <section className="workspace-panel"><div className="workspace-panel-heading"><div><span className="panel-kicker">Conexión y preferencias</span><h2>Configuración</h2><p>Última sincronización: {new Date(syncedAt).toLocaleString('es-MX')}</p></div><span className="connection-badge"><CheckCircle2 size={14} />{connection.status === 'connected' ? El archivo seguirá en tu Drive.')) void run(onDisconnect); }}><X size={15} />Desconectar</button></div><div className="settings-card"><h3>Sesión y datos</h3><p>Cerrar sesión solo termina Firebase Authentication. Las eliminaciones son lógicas en Google Sheets.</p><button type="button" className="toolbar-button danger-button" disabled={busy} onClick={() => { if (window.confirm('Se marcarán como eliminados todos los movimientos activos.')) void run(onDeleteAll); }}><Trash2 size={15} />Eliminar movimientos</button><button type="button" className="toolbar-button" disabled={busy} onClick={() => void run(onSignOut)}><LogOut size={15} />Cerrar sesión</button></div></div></section>;
}

export function Dashboard() {
  const navigate = useNavigate();
  const [user, setUser] = useState<User | null>(null);
  const [authReady, setAuthReady] = useState(false);
  const [connection, setConnection] = useState<GoogleConnection>();
  const [snapshot, setSnapshot] = useState<FinancialSnapshot>();
  const [view, setView] = useState<WorkspaceView>('dashboard');
  const [editing, setEditing] = useState<Transaction>();
  const [isEditorOpen, setEditorOpen] = useState(false);
  const [isMobileMenuOpen, setMobileMenuOpen] = useState(false);
  const [loadingConnection, setLoadingConnection] = useState(false);
  const [syncing, setSyncing] = useState(false);
  const [error, setError] = useState<string>();
  const [toast, setToast] = useState('');
  const [insights, setInsights] = useState<AiInsightsResponse>();
  const initialising = useRef(false);

  const showToast = useCallback((message: string) => { setToast(message); window.setTimeout(() => setToast(''), 3500); }, []);

  const refreshSnapshot = useCallback(async (notice = false) => {
    setSyncing(true);
    try {
      const next = await (notice ? syncFinancialSnapshot() : getFinancialSnapshot());
      setSnapshot(next);
      setInsights(buildDeterministicInsights(next.analytics, next.transactions));
      setError(undefined);
      if (notice) showToast('Sincronización terminada.');
      return next;
    } catch (caught) {
      const apiError = caught instanceof FinancialApiError ? caught : undefined;
      if (apiError?.code === 'GOOGLE_REAUTH_REQUIRED') setConnection((current) => ({ ...current, status: 'reauth_required' }));
      setError(errorMessage(caught));
      throw caught;
    } finally { setSyncing(false); }
  }, [showToast]);

  const ensureStorage = useCallback(async () => {
    setLoadingConnection(true); setError(undefined);
    try {
      const nextConnection = await ensureFinancialStorage();
      setConnection(nextConnection);
      await refreshSnapshot();
      showToast('Tu archivo financiero está listo.');
    } catch (caught) { setError(errorMessage(caught)); throw caught; } finally { setLoadingConnection(false); }
  }, [refreshSnapshot, showToast]);

  const loadConnection = useCallback(async () => {
    setLoadingConnection(true); setError(undefined);
    try {
      const nextConnection = await getConnection();
      setConnection(nextConnection);
      if (nextConnection.status === 'connected') await refreshSnapshot();
      if (nextConnection.status === 'authorized') await ensureStorage();
    } catch (caught) { setError(errorMessage(caught)); } finally { setLoadingConnection(false); }
  }, [ensureStorage, refreshSnapshot]);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (nextUser) => {
      setUser(nextUser); setAuthReady(true);
      if (!nextUser) { navigate('/auth', { replace: true }); return; }
      if (!initialising.current) { initialising.current = true; void loadConnection(); }
    });
    return unsubscribe;
  }, [loadConnection, navigate]);

  useEffect(() => {
    const onFocus = () => { if (connection?.status === 'connected') void refreshSnapshot(); };
    window.addEventListener('focus', onFocus);
    return () => window.removeEventListener('focus', onFocus);
  }, [connection?.status, refreshSnapshot]);

  const startAuthorization = useCallback(async () => {
    setLoadingConnection(true); setError(undefined);
    try { const { authorizationUrl } = await startGoogleAuthorization(); window.location.assign(authorizationUrl); } catch (caught) { setError(errorMessage(caught)); throw caught; } finally { setLoadingConnection(false); }
  }, []);

  const saveTransaction = async (payload: TransactionPayload, idempotencyKey: string) => {
    if (editing) await patchTransaction(editing.id, editing.updatedAt, payload);
    else await createTransaction(payload, idempotencyKey);
    await refreshSnapshot(true);
    showToast(editing ? 'Movimiento actualizado en Google Sheets.' : 'Movimiento guardado en Google Sheets.');
  };

  const removeTransaction = async (transaction: Transaction) => {
    if (!window.confirm(`¿Marcar “${transaction.description}” como eliminado?`)) return;
    try { await deleteTransaction(transaction.id, transaction.updatedAt); await refreshSnapshot(true); showToast('Movimiento eliminado lógicamente.'); } catch (caught) { if (caught instanceof FinancialApiError && caught.code === 'CONFLICT') void refreshSnapshot(true).catch(() => undefined); showToast(errorMessage(caught)); }
  };

  const removeAll = async () => { await deleteAllTransactions(); await refreshSnapshot(true); showToast('Los movimientos se marcaron como eliminados.'); };
  const saveBudgetForCategory = async (id: string, categoryId: string, amount: number, expectedUpdatedAt?: string) => { const range = monthBounds(); await saveBudget(id, { categoryId, amount, period: 'Mensual', ...range, active: true }, expectedUpdatedAt); await refreshSnapshot(true); };
  const saveSheetPreferences = async (preferences: Partial<FinancialPreferences>, expectedUpdatedAt: string) => { await savePreferences(preferences, expectedUpdatedAt); await refreshSnapshot(true); };
  const reconnect = async () => { await startAuthorization(); };
  const disconnect = async () => { await disconnectGoogle(); setConnection({ status: 'not_connected' }); setSnapshot(undefined); setInsights(undefined); showToast('La conexión se eliminó. Tu archivo sigue en Google Drive.'); };
  const signOutUser = async () => { await signOut(auth); navigate('/auth', { replace: true }); };

  const currentInsights = insights ?? (snapshot ? buildDeterministicInsights(snapshot.analytics, snapshot.transactions) : undefined);
  const openEditor = (transaction?: Transaction) => { setEditing(transaction); setEditorOpen(true); };
  const closeEditor = () => { setEditorOpen(false); setEditing(undefined); };
  const selectView = (next: WorkspaceView) => { setView(next); setMobileMenuOpen(false); if (next === 'insights' && snapshot) { void getInsights().then(setInsights).catch(() => undefined); } };

  if (!authReady) return <LoadingScreen message="Comprobando tu sesión…" />;
  if (!user) return null;
  if (!connection || (loadingConnection && !snapshot)) return <LoadingScreen message="Preparando tu conexión segura…" />;
  if (connection.status !== 'connected' || !snapshot) return <GoogleStorageOnboarding connection={connection} busy={loadingConnection} error={error} onAuthorize={startAuthorization} onEnsure={ensureStorage} />;

  const greeting = displayName(user);
  const renderView = () => {
    if (view === 'transactions') return <TransactionsPanel transactions={snapshot.transactions} currency={snapshot.preferences.currency} onAdd={() => openEditor()} onEdit={openEditor} onDelete={removeTransaction} />;
    if (view === 'income') return <TransactionsPanel transactions={snapshot.transactions} currency={snapshot.preferences.currency} filter="income" onAdd={() => openEditor()} onEdit={openEditor} onDelete={removeTransaction} />;
    if (view === 'expense') return <TransactionsPanel transactions={snapshot.transactions} currency={snapshot.preferences.currency} filter="expense" onAdd={() => openEditor()} onEdit={openEditor} onDelete={removeTransaction} />;
    if (view === 'categories') return <CategoriesPanel categories={snapshot.categories} />;
    if (view === 'budgets' || view === 'goals') return <BudgetPanel categories={snapshot.categories} budgets={snapshot.budgets} preferences={snapshot.preferences} onSave={saveBudgetForCategory} />;
    if (view === 'settings' || view === 'reports') return <SettingsPanel connection={connection} preferences={snapshot.preferences} syncedAt={snapshot.syncedAt} onSync={() => refreshSnapshot(true)} onSavePreferences={saveSheetPreferences} onReconnect={reconnect} onDisconnect={disconnect} onSignOut={signOutUser} onDeleteAll={removeAll} />;
    if (view === 'insights' && currentInsights) return <section className="workspace-panel insights-workspace"><div className="workspace-panel-heading"><div><span className="panel-kicker">Métricas agregadas</span><h2>Insights</h2><p>{currentInsights.summary}</p></div><button type="button" className="toolbar-button" onClick={() => void getInsights().then(setInsights).catch((caught) => showToast(errorMessage(caught)))}><RefreshCw size={15} />Actualizar</button></div><InsightsCard data={currentInsights} showAll onShowAll={() => undefined} /><div className="recommendation-list">{currentInsights.recommendations.map((recommendation) => <p key={recommendation}><Check size={14} />{recommendation}</p>)}</div></section>;
    return <DashboardOverview snapshot={snapshot} insights={currentInsights!} onAdd={() => openEditor()} onShowIssues={() => selectView('settings')} />;
  };

  return <div className="cuantly-app"><aside className="sidebar"><div className="brand sidebar-brand"><CuantlyMark /><div><strong>Cuantly</strong><small>Controla. Analiza. Decide.</small></div></div><nav className="side-nav" aria-label="Navegación principal">{navItems.map(({ id, label, icon: Icon }) => <button key={id} type="button" className={view === id || (id === 'goals' && view === 'budgets') ? 'is-active' : ''} onClick={() => selectView(id)}><Icon size={17} /><span>{label}</span></button>)}</nav><div className="sidebar-spacer" /><div className="support-card"><div className="support-illustration" aria-hidden="true"><div className="support-bot"><span /><span /><i /></div><Sparkles size={18} /></div><strong>Archivo conectado</strong><p>Tu información se sincroniza desde Google Sheets.</p><button type="button" onClick={() => openEditor()}><Plus size={14} />Nuevo movimiento</button></div><button type="button" className="collapse-button" onClick={() => selectView('settings')}><Settings size={16} />Configuración</button></aside><div className="app-shell"><div className="dashboard-column"><header className="desktop-topbar"><div className="desktop-greeting"><h1>¡Hola, {greeting}! <span>👋</span></h1><p>Así van tus finanzas registradas en tu archivo.</p></div><div className="topbar-controls"><span className="sync-indicator">{syncing ? <LoaderCircle className="spin" size={13} /> : <CheckCircle2 size={13} />} {syncing ? 'Sincronizando' : 'Sheets conectado'}</span><button type="button" className="toolbar-button compact" onClick={() => void refreshSnapshot(true)} disabled={syncing}><RefreshCw size={15} />Sincronizar</button><button className="user-button" type="button" onClick={() => selectView('settings')}><span className="avatar">{initials(user)}</span><span>{greeting}</span><ChevronDown size={14} /></button><button type="button" className="icon-button" onClick={() => void signOutUser()} aria-label="Cerrar sesión"><LogOut size={17} /></button></div></header><header className="mobile-topbar"><button type="button" className="mobile-menu-button" onClick={() => setMobileMenuOpen(true)} aria-label="Abrir menú"><Menu size={20} /></button><div className="brand mobile-brand"><CuantlyMark size={23} /><strong>Cuantly</strong></div><div className="mobile-top-actions"><button type="button" className="icon-button" onClick={() => void refreshSnapshot(true)} aria-label="Sincronizar"><RefreshCw size={17} /></button><span className="avatar">{initials(user)}</span></div></header>{isMobileMenuOpen && <div className="mobile-menu-scrim" onClick={() => setMobileMenuOpen(false)} />}<aside className={`mobile-drawer ${isMobileMenuOpen ? 'is-open' : ''}`} aria-label="Menú móvil"><div className="drawer-heading"><div className="brand"><CuantlyMark /><strong>Cuantly</strong></div><button type="button" onClick={() => setMobileMenuOpen(false)} aria-label="Cerrar menú"><X size={19} /></button></div>{navItems.map(({ id, label, icon: Icon }) => <button type="button" key={id} className={view === id ? 'is-active' : ''} onClick={() => selectView(id)}><Icon size={17} /><span>{label}</span></button>)}</aside><main className="dashboard-main"><div className="mobile-greeting"><h1>¡Hola, {greeting}! <span>👋</span></h1><p>Datos sincronizados desde Google Sheets.</p></div>{error && <p className="global-error" role="alert">{error}</p>}{renderView()}</main><nav className="mobile-bottom-nav" aria-label="Navegación móvil"><button type="button" className={view === 'dashboard' ? 'is-active' : ''} onClick={() => selectView('dashboard')}><House size={17} /><span>Dashboard</span></button><button type="button" className={view === 'transactions' ? 'is-active' : ''} onClick={() => selectView('transactions')}><ReceiptText size={17} /><span>Movimientos</span></button><button type="button" className="mobile-add" onClick={() => openEditor()} aria-label="Nuevo movimiento"><Plus size={25} /></button><button type="button" className={view === 'insights' ? 'is-active' : ''} onClick={() => selectView('insights')}><Lightbulb size={17} /><span>Insights</span></button><button type="button" className={view === 'settings' ? 'is-active' : ''} onClick={() => selectView('settings')}><MoreHorizontal size={18} /><span>Más</span></button></nav></div></div>{isEditorOpen && <TransactionEditor key={editing?.id ?? 'new'} transaction={editing} categories={snapshot.categories} onClose={closeEditor} onSave={saveTransaction} />}{toast && <div className="toast" role="status"><Check size={15} />{toast}</div>}</div>;
}
